import XCTest

@MainActor final class ReferenceUITests: XCTestCase {
    func testBareNativeTextField() throws { try audit([]) }
    func testMinimumHeightNativeTextField() throws { try audit(["--framed"]) }
    private func audit(_ arguments: [String]) throws {
        continueAfterFailure = false
        XCUIDevice.shared.orientation = .portrait
        let app = XCUIApplication()
        app.launchArguments = arguments
        app.launch()
        XCTAssertTrue(app.textFields["field"].waitForExistence(timeout: 15))
        let hierarchy = XCTAttachment(string: app.debugDescription)
        hierarchy.name = "Reference field hierarchy"; hierarchy.lifetime = .keepAlways; add(hierarchy)
        let capture = XCTAttachment(screenshot: XCUIScreen.main.screenshot())
        capture.name = "Reference field"; capture.lifetime = .keepAlways; add(capture)
        try app.performAccessibilityAudit(for: [.textClipped])
    }
}
